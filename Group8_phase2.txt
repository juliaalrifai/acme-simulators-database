
-- DDL - TABLE CREATION & CONSTRAINTS
-- ------------------------------------------------------------

DROP TABLE IF EXISTS TEST_EXECUTIONS;
DROP TABLE IF EXISTS PROJECT_DEPLOYED_VERSIONS;
DROP TABLE IF EXISTS CHANGE_LOGS;
DROP TABLE IF EXISTS APPROVAL_TASKS;
DROP TABLE IF EXISTS VERSION_ROWS;
DROP TABLE IF EXISTS DOCUMENT_VERSIONS;
DROP TABLE IF EXISTS DOCUMENTS;
DROP TABLE IF EXISTS PROJECT_OPTION_SELECTIONS;
DROP TABLE IF EXISTS PROJECTS;
DROP TABLE IF EXISTS PLATFORM_OPTIONS;
DROP TABLE IF EXISTS PLATFORMS;
DROP TABLE IF EXISTS PRODUCT_OPTIONS;
DROP TABLE IF EXISTS PRODUCTS;
DROP TABLE IF EXISTS USERS;

-- Create USERS Table
CREATE TABLE USERS (
    user_id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    role_name VARCHAR(30) NOT NULL CHECK (role_name IN ('Admin', 'Author', 'Reviewer', 'Tester', 'Client'))
);

-- Create PRODUCTS Table
CREATE TABLE PRODUCTS (
    product_id VARCHAR(20) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    family VARCHAR(50) NOT NULL,
    type VARCHAR(50) NOT NULL,
    standard_ver VARCHAR(20) NOT NULL,
    model_year INT NOT NULL,
    product_type VARCHAR(20) NOT NULL CHECK (product_type IN ('Civil', 'Military')),
    created_by INT NOT NULL,
    created_on DATE NOT NULL,
    status VARCHAR(20) NOT NULL CHECK (status IN ('Active', 'Deprecated', 'Archived')),
    FOREIGN KEY (created_by) REFERENCES USERS(user_id) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Create PRODUCT_OPTIONS Table
CREATE TABLE PRODUCT_OPTIONS (
    product_option_id INT PRIMARY KEY AUTO_INCREMENT,
    product_id VARCHAR(20) NOT NULL,
    option_name VARCHAR(50) NOT NULL,
    option_type VARCHAR(20) NOT NULL CHECK (option_type IN ('Enum', 'Boolean', 'Text')),
    valid_values VARCHAR(255) NOT NULL,
    FOREIGN KEY (product_id) REFERENCES PRODUCTS(product_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- Create PLATFORMS Table
CREATE TABLE PLATFORMS (
    platform_id VARCHAR(20) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    version VARCHAR(20) NOT NULL,
    platform_year INT NOT NULL,
    tech_level VARCHAR(20) NOT NULL CHECK (tech_level IN ('Low', 'Medium', 'High')),
    support_4d VARCHAR(10) NOT NULL CHECK (support_4d IN ('Yes', 'No')),
    platform_type VARCHAR(20) NOT NULL CHECK (platform_type IN ('Civil', 'Military')),
    created_by INT NOT NULL,
    created_on DATE NOT NULL,
    status VARCHAR(20) NOT NULL CHECK (status IN ('Active', 'Deprecated', 'Archived')),
    FOREIGN KEY (created_by) REFERENCES USERS(user_id) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Create PLATFORM_OPTIONS Table
CREATE TABLE PLATFORM_OPTIONS (
    platform_option_id INT PRIMARY KEY AUTO_INCREMENT,
    platform_id VARCHAR(20) NOT NULL,
    option_name VARCHAR(50) NOT NULL,
    option_type VARCHAR(20) NOT NULL CHECK (option_type IN ('Enum', 'Boolean', 'Text')),
    valid_values VARCHAR(255) NOT NULL,
    FOREIGN KEY (platform_id) REFERENCES PLATFORMS(platform_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- Create PROJECTS Table
CREATE TABLE PROJECTS (
    project_id VARCHAR(30) PRIMARY KEY,
    project_name VARCHAR(150) NOT NULL,
    client_name VARCHAR(100) NOT NULL,
    product_id VARCHAR(20) NOT NULL,
    platform_id VARCHAR(20) NOT NULL,
    start_date DATE NOT NULL,
    status VARCHAR(30) NOT NULL CHECK (status IN ('In Configuration', 'Active', 'Testing', 'Completed', 'Archived')),
    created_by INT NOT NULL,
    created_on DATE NOT NULL,
    last_updated DATE,
    FOREIGN KEY (product_id) REFERENCES PRODUCTS(product_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (platform_id) REFERENCES PLATFORMS(platform_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (created_by) REFERENCES USERS(user_id) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Create PROJECT_OPTION_SELECTIONS Table
CREATE TABLE PROJECT_OPTION_SELECTIONS (
    selection_id INT PRIMARY KEY AUTO_INCREMENT,
    project_id VARCHAR(30) NOT NULL,
    option_source VARCHAR(20) NOT NULL CHECK (option_source IN ('Product', 'Platform')),
    option_name VARCHAR(50) NOT NULL,
    selected_value VARCHAR(100) NOT NULL,
    FOREIGN KEY (project_id) REFERENCES PROJECTS(project_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- Create DOCUMENTS Table
CREATE TABLE DOCUMENTS (
    doc_id VARCHAR(20) PRIMARY KEY,
    doc_name VARCHAR(150) NOT NULL,
    description TEXT,
    target_type VARCHAR(20) NOT NULL CHECK (target_type IN ('Product', 'Platform')),
    product_id VARCHAR(20),
    platform_id VARCHAR(20),
    created_at DATE NOT NULL,
    created_by INT NOT NULL,
    is_locked INT DEFAULT 0 CHECK (is_locked IN (0, 1)),
    locked_by INT,
    FOREIGN KEY (product_id) REFERENCES PRODUCTS(product_id) ON DELETE SET NULL ON UPDATE CASCADE,
    FOREIGN KEY (platform_id) REFERENCES PLATFORMS(platform_id) ON DELETE SET NULL ON UPDATE CASCADE,
    FOREIGN KEY (created_by) REFERENCES USERS(user_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (locked_by) REFERENCES USERS(user_id) ON DELETE SET NULL ON UPDATE CASCADE
);

-- Create DOCUMENT_VERSIONS Table
CREATE TABLE DOCUMENT_VERSIONS (
    version_id INT PRIMARY KEY AUTO_INCREMENT,
    doc_id VARCHAR(20) NOT NULL,
    version_number INT NOT NULL,
    version_tag VARCHAR(30) NOT NULL,
    release_id VARCHAR(30) UNIQUE,
    status VARCHAR(20) NOT NULL CHECK (status IN ('Draft', 'Pending Approval', 'Approved', 'Rejected')),
    created_by INT NOT NULL,
    created_at DATE NOT NULL,
    FOREIGN KEY (doc_id) REFERENCES DOCUMENTS(doc_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (created_by) REFERENCES USERS(user_id) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Create VERSION_ROWS Table
CREATE TABLE VERSION_ROWS (
    row_pk INT PRIMARY KEY AUTO_INCREMENT,
    version_id INT NOT NULL,
    row_id VARCHAR(20) NOT NULL,
    row_order INT NOT NULL,
    row_type VARCHAR(20) NOT NULL CHECK (row_type IN ('H1', 'H2', 'H3', 'TEXT', 'ACTION')),
    content TEXT NOT NULL,
    observation TEXT,
    condition_expr VARCHAR(255),
    comment TEXT,
    FOREIGN KEY (version_id) REFERENCES DOCUMENT_VERSIONS(version_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- Create APPROVAL_TASKS Table
CREATE TABLE APPROVAL_TASKS (
    task_id INT PRIMARY KEY AUTO_INCREMENT,
    version_id INT NOT NULL,
    reviewer_id INT NOT NULL,
    approval_status VARCHAR(20) NOT NULL CHECK (approval_status IN ('Pending', 'Approved', 'Rejected')),
    review_comments TEXT,
    reviewed_at DATE,
    FOREIGN KEY (version_id) REFERENCES DOCUMENT_VERSIONS(version_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (reviewer_id) REFERENCES USERS(user_id) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Create CHANGE_LOGS Table
CREATE TABLE CHANGE_LOGS (
    log_id INT PRIMARY KEY AUTO_INCREMENT,
    version_id INT NOT NULL,
    operation_type VARCHAR(20) NOT NULL CHECK (operation_type IN ('ADD_ROW', 'EDIT_ROW', 'DELETE_ROW', 'REORDER')),
    row_id VARCHAR(20) NOT NULL,
    performed_by INT NOT NULL,
    performed_at DATE NOT NULL,
    details TEXT,
    FOREIGN KEY (version_id) REFERENCES DOCUMENT_VERSIONS(version_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (performed_by) REFERENCES USERS(user_id) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Create PROJECT_DEPLOYED_VERSIONS Table
CREATE TABLE PROJECT_DEPLOYED_VERSIONS (
    project_id VARCHAR(30) NOT NULL,
    version_id INT NOT NULL,
    PRIMARY KEY (project_id, version_id),
    FOREIGN KEY (project_id) REFERENCES PROJECTS(project_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (version_id) REFERENCES DOCUMENT_VERSIONS(version_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- Create TEST_EXECUTIONS Table
CREATE TABLE TEST_EXECUTIONS (
    test_id INT PRIMARY KEY AUTO_INCREMENT,
    project_id VARCHAR(30) NOT NULL,
    version_id INT NOT NULL,
    row_pk INT NOT NULL,
    phase VARCHAR(20) NOT NULL CHECK (phase IN ('Company', 'Client')),
    step_description TEXT NOT NULL,
    expected_observation TEXT,
    result VARCHAR(10) NOT NULL CHECK (result IN ('Pass', 'Fail', 'N/A')),
    executed_by INT NOT NULL,
    execution_date DATE NOT NULL,
    FOREIGN KEY (project_id) REFERENCES PROJECTS(project_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (version_id) REFERENCES DOCUMENT_VERSIONS(version_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (row_pk) REFERENCES VERSION_ROWS(row_pk) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (executed_by) REFERENCES USERS(user_id) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Query 1: Simple SELECT - Retrieve all active Civil products
-- Objective: Filter active product offerings in the civil driving simulation space.
SELECT product_id, name, family, model_year, standard_ver 
FROM PRODUCTS 
WHERE product_type = 'Civil' AND status = 'Active';


-- Query 2: Filtering with Range & Sorting - High-technology platforms
-- Objective: Identify simulator platforms released in or after 2021 with High tech level.
SELECT platform_id, name, version, platform_year, support_4d 
FROM PLATFORMS 
WHERE platform_year >= 2021 AND tech_level = 'High'
ORDER BY platform_year DESC;


-- Query 3: Multi-Table Join - Complete list of projects with Product and Platform names
-- Objective: Executive listing showing client projects along with assigned vehicle models and platforms.
SELECT p.project_id, p.project_name, p.client_name, 
       prod.name AS assigned_product, 
       plat.name AS assigned_platform, 
       p.status
FROM PROJECTS p
JOIN PRODUCTS prod ON p.product_id = prod.product_id
JOIN PLATFORMS plat ON p.platform_id = plat.platform_id;


-- Query 4: Aggregate Function - Count of document versions by release status
-- Objective: Summarize version counts across life cycle states (Approved, Draft, Rejected).
SELECT status, COUNT(version_id) AS total_versions
FROM DOCUMENT_VERSIONS
GROUP BY status;


-- Query 5: INNER JOIN with Filtering - Detailed Project Handbook Configuration Options
-- Objective: Retrieve all configured option selections for project 'PRO_2025_0731'.
SELECT pos.project_id, p.project_name, pos.option_source, pos.option_name, pos.selected_value
FROM PROJECT_OPTION_SELECTIONS pos
JOIN PROJECTS p ON pos.project_id = p.project_id
WHERE pos.project_id = 'PRO_2025_0731'
ORDER BY pos.option_source, pos.option_name;


-- Query 6: Document Versioning Lineage Join - Documents and their Approved Release IDs
-- Objective: Show all documents, their published release IDs, and authors.
SELECT d.doc_id, d.doc_name, dv.version_tag, dv.release_id, u.full_name AS author_name
FROM DOCUMENTS d
JOIN DOCUMENT_VERSIONS dv ON d.doc_id = dv.doc_id
JOIN USERS u ON dv.created_by = u.user_id
WHERE dv.status = 'Approved';


-- Query 7: Conditional Handbook Generator - Filter ENG-01 Document Rows for Hybrid + 4D Setup
-- Objective: Dynamic evaluation simulator generating printable handbook content for a Hybrid, 4D project.
SELECT r.row_order, r.row_id, r.row_type, r.content, r.observation, r.condition_expr
FROM VERSION_ROWS r
JOIN DOCUMENT_VERSIONS dv ON r.version_id = dv.version_id
WHERE dv.release_id = 'R3-ENG-01'
  AND (r.condition_expr IS NULL 
       OR r.condition_expr = 'Engine=''hybrid''' 
       OR r.condition_expr = '$4D=true$')
ORDER BY r.row_order ASC;


-- Query 8: Grouping & Aggregate - Number of Documents authored per User
-- Objective: Identify author productivity across the engineering documentation team.
SELECT u.user_id, u.full_name, COUNT(d.doc_id) AS documents_authored
FROM USERS u
LEFT JOIN DOCUMENTS d ON u.user_id = d.created_by
WHERE u.role_name IN ('Author', 'Admin')
GROUP BY u.user_id, u.full_name;


-- Query 9: Subquery in WHERE - Documents that have no approved release
-- Objective: Uncover stagnant or unapproved documents requiring review attention.
SELECT doc_id, doc_name, created_at 
FROM DOCUMENTS 
WHERE doc_id NOT IN (
    SELECT DISTINCT doc_id 
    FROM DOCUMENT_VERSIONS 
    WHERE status = 'Approved'
);


-- Query 10: Multi-Table Audit Join - Line-by-line Change Log for Document Versioning
-- Objective: Trace change activity for Version 3 of ENG-01 including user signatures.
SELECT cl.log_id, dv.version_tag, cl.operation_type, cl.row_id, cl.details, u.full_name AS modifier, cl.performed_at
FROM CHANGE_LOGS cl
JOIN DOCUMENT_VERSIONS dv ON cl.version_id = dv.version_id
JOIN USERS u ON cl.performed_by = u.user_id
WHERE dv.version_tag = 'V3-ENG-01'
ORDER BY cl.performed_at DESC;


-- Query 11: Test Results Summary by Phase for Project 'SIM-CRV-003'
-- Objective: Calculate total passes, fails, and N/A steps split between Company and Client testing.
SELECT phase,
       SUM(CASE WHEN result = 'Pass' THEN 1 ELSE 0 END) AS pass_count,
       SUM(CASE WHEN result = 'Fail' THEN 1 ELSE 0 END) AS fail_count,
       SUM(CASE WHEN result = 'N/A' THEN 1 ELSE 0 END) AS na_count,
       COUNT(*) AS total_steps_executed
FROM TEST_EXECUTIONS
WHERE project_id = 'SIM-CRV-003'
GROUP BY phase;


-- Query 12: Subquery with Aggregate Comparison - Projects with above-average option count
-- Objective: Find highly customized projects that have configured more options than average.
SELECT project_id, COUNT(selection_id) AS configured_options_count
FROM PROJECT_OPTION_SELECTIONS
GROUP BY project_id
HAVING COUNT(selection_id) > (
    SELECT AVG(opt_count) 
    FROM (
        SELECT COUNT(selection_id) AS opt_count 
        FROM PROJECT_OPTION_SELECTIONS 
        GROUP BY project_id
    ) AS sub
);


-- Query 13: LEFT JOIN - Products and their assigned Document counts
-- Objective: Audit coverage to ensure every vehicle model has associated documentation.
SELECT p.product_id, p.name AS product_name, COUNT(d.doc_id) AS document_count
FROM PRODUCTS p
LEFT JOIN DOCUMENTS d ON p.product_id = d.product_id
GROUP BY p.product_id, p.name;


-- Query 14: Approval Workflow Status Matrix
-- Objective: Comprehensive review task status report listing reviewer, version tag, and comments.
SELECT t.task_id, dv.version_tag, u.full_name AS reviewer, t.approval_status, t.review_comments, t.reviewed_at
FROM APPROVAL_TASKS t
JOIN DOCUMENT_VERSIONS dv ON t.version_id = dv.version_id
JOIN USERS u ON t.reviewer_id = u.user_id;


-- Query 15: Deployed Release Versions across Active Projects
-- Objective: Map client projects to exact document versions currently deployed in production.
SELECT pdv.project_id, pr.client_name, d.doc_name, dv.version_tag, dv.release_id
FROM PROJECT_DEPLOYED_VERSIONS pdv
JOIN PROJECTS pr ON pdv.project_id = pr.project_id
JOIN DOCUMENT_VERSIONS dv ON pdv.version_id = dv.version_id
JOIN DOCUMENTS d ON dv.doc_id = d.doc_id;


-- Query 16: Failure Analysis Report - Failed Client Test Executions
-- Objective: Identify critical failure points during client acceptance testing for quick remediation.
SELECT te.test_id, te.project_id, te.step_description, te.expected_observation, u.full_name AS tester
FROM TEST_EXECUTIONS te
JOIN USERS u ON te.executed_by = u.user_id
WHERE te.phase = 'Client' AND te.result = 'Fail';


-- Query 17: Union Query - Complete Document & Project Lifecycle Timeline
-- Objective: Consolidated chronological feed of major system milestones.
SELECT created_on AS activity_date, 'Project Created' AS activity_type, project_name AS entity_name
FROM PROJECTS
UNION ALL
SELECT created_at AS activity_date, 'Document Uploaded' AS activity_type, doc_name AS entity_name
FROM DOCUMENTS
UNION ALL
SELECT execution_date AS activity_date, 'Test Conducted' AS activity_type, step_description AS entity_name
FROM TEST_EXECUTIONS
ORDER BY activity_date DESC;


-- Query 18: Self-Join - Find Authors sharing the exact same role as 'John Smith'
-- Objective: Locate team peers with identical system privileges.
SELECT u1.user_id, u1.full_name, u1.email, u1.role_name
FROM USERS u1
JOIN USERS u2 ON u1.role_name = u2.role_name
WHERE u2.username = 'jsmith' AND u1.username <> 'jsmith';


-- Query 19: Correlated Subquery - Latest Approved Version per Document
-- Objective: Fetch the highest approved version number for each document in the catalog.
SELECT v1.doc_id, v1.version_tag, v1.release_id, v1.version_number
FROM DOCUMENT_VERSIONS v1
WHERE v1.status = 'Approved'
  AND v1.version_number = (
      SELECT MAX(v2.version_number)
      FROM DOCUMENT_VERSIONS v2
      WHERE v2.doc_id = v1.doc_id AND v2.status = 'Approved'
  );


-- Query 20: Complex Cross-Domain Audit Query - Failed Step Lineage
-- Objective: Full traceability report connecting project, vehicle, platform, failing row, and tester.
SELECT pr.project_id, 
       pr.client_name, 
       prod.name AS vehicle_model, 
       plat.name AS platform_model, 
       dv.release_id, 
       vr.row_id, 
       te.step_description, 
       te.result, 
       u.full_name AS executed_by_user
FROM TEST_EXECUTIONS te
JOIN PROJECTS pr ON te.project_id = pr.project_id
JOIN PRODUCTS prod ON pr.product_id = prod.product_id
JOIN PLATFORMS plat ON pr.platform_id = plat.platform_id
JOIN VERSION_ROWS vr ON te.row_pk = vr.row_pk
JOIN DOCUMENT_VERSIONS dv ON vr.version_id = dv.version_id
JOIN USERS u ON te.executed_by = u.user_id
WHERE te.result = 'Fail';
